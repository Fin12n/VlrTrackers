<?php

$curl = curl_init();

curl_setopt_array($curl, [
  CURLOPT_URL => "https://api.henrikdev.xyz/valorant/v1/account/{name}/{tag}",

  ],

$err = curl_error($curl);

  echo "cURL Error #:" . $err;
} else {
  echo $response;
}
```
